extends Node
## Project current sprite silhouettes onto the floor without changing color art.
## A quarter-size GPU target keeps soft shadows out of the native sprite pass;
## no image readbacks, generated atlases or persistent animation-frame cache.
const SHADER = preload("ground_shadow.gdshader")
const RESOLUTION := 0.25
const OPACITY := 0.22
const PROJECTION_X := Vector2(1.0, 0.0)
# Sprite height is negative Y: this projects it rightward and upward. Give
# raised decks enough reach that their shadow clears their own color sprite.
const PROJECTION_Y := Vector2(-0.55, 0.35)
var viewport: SubViewport
var sprites: Array[Sprite2D] = []
var count := 0

func _ready() -> void:
	viewport = SubViewport.new()
	viewport.disable_3d = true
	viewport.transparent_bg = true
	viewport.gui_disable_input = true
	viewport.render_target_clear_mode = SubViewport.CLEAR_MODE_ALWAYS
	viewport.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	add_child(viewport)

func begin(dimensions: Vector2) -> void:
	set_active(true)
	viewport.size = Vector2i(maxi(1, ceili(dimensions.x * RESOLUTION)), maxi(1, ceili(dimensions.y * RESOLUTION)))
	count = 0

func set_active(active: bool) -> void:
	viewport.render_target_update_mode = SubViewport.UPDATE_ALWAYS if active else SubViewport.UPDATE_DISABLED

func add_silhouette(frame: Dictionary, destination: Rect2, ground: Vector2, strength: float) -> void:
	var sprite: Sprite2D
	if count == sprites.size():
		sprite = Sprite2D.new()
		sprite.centered = false
		sprite.region_enabled = true
		sprite.texture_filter = CanvasItem.TEXTURE_FILTER_LINEAR
		var shader_material := ShaderMaterial.new()
		shader_material.shader = SHADER
		sprite.material = shader_material
		viewport.add_child(sprite)
		sprites.append(sprite)
	else:
		sprite = sprites[count]
	count += 1
	var texture: Texture2D = frame.texture
	var region: Rect2 = frame.region
	var texture_size := texture.get_size()
	var source_scale := destination.size / region.size
	# Room for the blur outside the original alpha/frame boundary.
	var padding := Vector2(3.0, 9.0)
	var expanded := region.grow_individual(padding.x, padding.y, padding.x, padding.y)
	sprite.texture = texture
	sprite.region_rect = expanded
	sprite.offset = (destination.position - ground) / source_scale - padding
	sprite.transform = Transform2D(PROJECTION_X * source_scale.x * RESOLUTION, PROJECTION_Y * source_scale.y * RESOLUTION, ground * RESOLUTION)
	sprite.modulate = Color(1, 1, 1, OPACITY * strength)
	var shader_material: ShaderMaterial = sprite.material
	shader_material.set_shader_parameter("frame_uv", Vector4(region.position.x / texture_size.x, region.position.y / texture_size.y, region.end.x / texture_size.x, region.end.y / texture_size.y))
	shader_material.set_shader_parameter("softness", Vector2(2.0, 7.0) / texture_size)
	sprite.show()

func finish() -> void:
	# Release atlas references when clips/facings change or objects leave view.
	for index in range(count, sprites.size()):
		sprites[index].texture = null
		sprites[index].hide()

func texture() -> Texture2D:
	return viewport.get_texture()
